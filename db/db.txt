CREATE TABLE `lotto` (
  `idlotto` int NOT NULL,
  `data_produzione` date NOT NULL,
  `num_bottiglie` int NOT NULL,
  PRIMARY KEY (`idlotto`),
  UNIQUE KEY `idlotto_UNIQUE` (`idlotto`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci


CREATE TABLE `tipologia` (
  `idtipologia` int NOT NULL AUTO_INCREMENT,
  `nome` varchar(45) NOT NULL,
  PRIMARY KEY (`idtipologia`)
) ENGINE=InnoDB AUTO_INCREMENT=12 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci

CREATE TABLE `tipologia_lotto` (
  `id_tipologia` int NOT NULL,
  `id_lotto` int NOT NULL,
  `numBott` int NOT NULL,
  `id` int NOT NULL AUTO_INCREMENT,
  PRIMARY KEY (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=28 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci

CREATE TABLE `vendita` (
  `idvendita` int NOT NULL AUTO_INCREMENT,
  `numBottiglie` int NOT NULL,
  `effettuata_programmata` enum('programmata','effettuata') DEFAULT NULL,
  `data_vendita` datetime NOT NULL,
  `id_combo` int NOT NULL,
  `idLotto` int NOT NULL,
  `cliente` enum('privato','pubblico') NOT NULL DEFAULT 'pubblico',
  `note` text,
  PRIMARY KEY (`idvendita`),
  KEY `id_combo_idx` (`id_combo`)
) ENGINE=InnoDB AUTO_INCREMENT=60 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci
